# 整体的项目结构规划结构

electron-app/
├── src/
│   ├── main/                    # 主进程代码 (TypeScript)
│   │   ├── main.ts             # 主进程入口
│   │   ├── windowManager.ts    # 窗口管理
│   │   ├── menu.ts             # 菜单管理
│   │   └── ipc/                # IPC 处理器
│   │       ├── fileIpc.ts	#
│   │       └── appIpc.ts 	# 
│   ├── renderer/               # 渲染进程代码
│   │   ├── assets/             # 静态资源
│   │   │   ├── styles/
│   │   │   │   ├── main.css
│   │   │   │   └── components.css
│   │   │   └── images/
│   │   ├── components/         # UI 组件
│   │   │   ├── Button/
│   │   │   │   ├── Button.tsx
│   │   │   │   └── Button.css
│   │   │   └── Header/
│   │   │       ├── Header.tsx
│   │   │       └── Header.css
│   │   ├── pages/              # 页面组件
│   │   │   ├── Home/
│   │   │   │   ├── Home.tsx
│   │   │   │   └── Home.css
│   │   │   └── Settings/
│   │   │       ├── Settings.tsx
│   │   │       └── Settings.css
│   │   ├── utils/              # 工具函数
│   │   │   ├── api.ts
│   │   │   └── helpers.ts
│   │   ├── App.tsx             # 根组件
│   │   └── main.tsx            # 渲染进程入口
│   ├── preload/                # 预加载脚本
│   │   ├── preload.ts
│   │   └── types.ts            # TypeScript 类型定义
│   └── shared/                 # 共享代码
│       ├── types/              # 共享类型定义
│       │   ├── electron.d.ts
│       │   └── api.d.ts
│       └── constants.ts        # 共享常量
├── dist/                       # 构建输出
├── build/                      # 构建资源
│   ├── icons/                  # 应用图标
│   └── scripts/                # 构建脚本
├── public/                     # 公共静态资源
├── vite.config.ts              # Vite 配置
├── electron-builder.json       # 打包配置
├── package.json
├── tsconfig.json
├── tsconfig.main.json          # 主进程 TS 配置
├── tsconfig.renderer.json      # 渲染进程 TS 配置
└── tsconfig.preload.json       # 预加载脚本 TS 配置
